"""
视频清理模块

删除已完成转录的视频文件，释放磁盘空间。

用法:
    python -m src.cleaner
    python -m src.cleaner --dry-run
"""

import sys
from pathlib import Path

from .config import load_config

VIDEO_EXTENSIONS = (".mp4", ".mkv", ".flv", ".avi", ".webm", ".mov")


def run(dry_run: bool = False):
    """清理已转录的视频"""
    config = load_config()
    video_dir = Path(config["paths"]["video_dir"])
    transcript_dir = Path(config["paths"]["transcript_dir"])

    transcribed_stems = {f.stem for f in transcript_dir.glob("*.txt")}

    to_delete = []
    for vf in video_dir.iterdir():
        if vf.suffix.lower() in VIDEO_EXTENSIONS and vf.stem in transcribed_stems:
            to_delete.append(vf)

    if not to_delete:
        print("[提示] 没有可清理的视频")
        return

    total_mb = sum(f.stat().st_size for f in to_delete) / 1024 / 1024
    print(f"[清理] {len(to_delete)} 个视频，共 {total_mb:.1f} MB\n")

    for vf in sorted(to_delete):
        size_mb = vf.stat().st_size / 1024 / 1024
        if dry_run:
            print(f"  [预览] {vf.name}  ({size_mb:.1f} MB)")
        else:
            vf.unlink()
            print(f"  [删除] {vf.name}  ({size_mb:.1f} MB)")

    if dry_run:
        print(f"\n[预览] 未实际删除，去掉 --dry-run 执行")
    else:
        print(f"\n[完成] 释放 {total_mb:.1f} MB")


if __name__ == "__main__":
    dry = "--dry-run" in sys.argv
    run(dry_run=dry)
